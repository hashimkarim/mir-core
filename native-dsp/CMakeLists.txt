cmake_minimum_required(VERSION 3.22)
project(mir_dsp VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MIR_DSP_JNI "Build the Android/JVM interface" OFF)
option(MIR_DSP_TESTS "Build host validation tools" ON)

# Same unmodified libsoxr sources as Python-SoXR 0.5.0.post1 in MIR.
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(soxr_source
    URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/soxr-0.5.0.post1.tar.gz"
    URL_HASH SHA256=7092b9f3e8a416044e1fa138c8172520757179763b85dc53aa9504f4813cff73
    SOURCE_SUBDIR libsoxr)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_OPENMP OFF CACHE BOOL "" FORCE)
set(WITH_LSR_BINDINGS OFF CACHE BOOL "" FORCE)
set(WITH_VR32 OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(soxr_source)
# Android packages unversioned shared libraries and supports 16 KiB pages.
if(ANDROID)
    set_target_properties(soxr PROPERTIES VERSION "" SOVERSION "")
    target_link_options(soxr PRIVATE "-Wl,-z,max-page-size=16384")
endif()

add_library(mir_dsp SHARED src/resampler.cpp)
target_include_directories(mir_dsp PUBLIC include PRIVATE "${soxr_source_SOURCE_DIR}/libsoxr/src")
target_link_libraries(mir_dsp PRIVATE soxr)
if(ANDROID)
    target_link_options(mir_dsp PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(MIR_DSP_JNI)
    if(NOT ANDROID)
        find_package(JNI REQUIRED)
        target_include_directories(mir_dsp PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
    target_sources(mir_dsp PRIVATE src/resampler_jni.cpp)
endif()
if(MIR_DSP_TESTS AND NOT ANDROID)
    enable_testing()
    add_executable(mir_resampler_replay tests/resampler_replay.cpp)
    target_link_libraries(mir_resampler_replay PRIVATE mir_dsp)
endif()
install(TARGETS mir_dsp LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
